/* REGION FR-DE: retail ships a real two-language selection screen here (the
   base build only clears frontEnd.language).  The shape names and the psh name
   are the regional image's own rodata block at 0x80012818; the two language
   codes are 1 (yDEU) and 2 (yFRA).
   [INFERRED] identities: the 48-byte stack blob handed to creatememclass, and
   the two RegionalLanguageShapeMemory* globals it takes its budget from. */
struct LanguageMemClass {
  char data[48];
};

extern "C" void creatememclass(...);
extern "C" void setmemclass(int);
extern "C" int VSync(int);
extern int GetPSXPadValue(int value, int player);
extern int RegionalLanguageShapeMemory;
extern int RegionalLanguageShapeMemorySize;

void DoLanguageScreen(void)

{
  Front_InitialMemCardCheck();
  if ((u_char)frontEnd.language == 0xff) {   /* sentinel: -1 == 0xFF as unsigned byte (this build's char is unsigned -> lbu;bne 0xFF) */
    char *shapeNames[7] = {
      "back", "yDEU", "yFRA", "bDEU", "bFRA", "lite", "dark"
    };
    shapetbl *shapes[7];
    u_char languages[2] = {1, 2};
    LanguageMemClass memclass;
    char *shapeFile;
    int selected;
    int flash;
    int key;
    int i;
    int x;
    int shapeOffset;
    shapetbl **shapeBase;

    selected = 0;
    creatememclass(&memclass,RegionalLanguageShapeMemory,
                   RegionalLanguageShapeMemorySize,"lang.psh");
    shapeFile = (char *)loadshapeadr(&memclass,0);
    setmemclass(0);
    i = 0;
    shapeBase = shapes;
    for (; i < 7; i++) {
      *shapeBase++ = (shapetbl *)locateshapez(shapeFile,shapeNames[i]);
    }
    flash = 0;
    Quick_DD(1,1,1);
    settrans(0);
    movfxya(shapes[0],flash,flash);
    shapeBase = shapes;
    key = flash;
    while ((key != 2) && (key != 0x2000)) {
      flash++;
      if (flash >= 2) {
        flash = 0;
      }
      settrans(1);
      i = 0;
      shapeOffset = 4;
      x = 0xe2 - shapes[5]->width;
      VSync(0);
      for (; i < 2; i++) {
        int markerOffset;
        int centerOffset;
        int selectedShapeOffset;

        markerOffset = 24;
        if (selected == i) {
          markerOffset = 20;
        }
        movfxya(*(shapetbl **)((char *)shapeBase + markerOffset),x,0xb4);
        centerOffset = ((short)shapes[5]->width >> 1) -
                       ((short)(*(shapetbl **)((char *)shapeBase + shapeOffset))->width >> 1);
        selectedShapeOffset = (i + 3) * 4;
        if (selected == i) {
          selectedShapeOffset = shapeOffset;
        }
        movfxya(*(shapetbl **)((char *)shapeBase + selectedShapeOffset),x + centerOffset,
                (languages[i] == 3) ? 0xb8 : 0xb9);
        shapeOffset += 4;
        x += 0x3c + shapes[5]->width;
      }
      key = GetPSXPadValue(0,-1);
      if (key == 0) {
        key = GetPSXPadValue(1,-1);
      }
      if (key == 0x800) {
        selected--;
      }
      if (key == 0x1000) {
        selected++;
      }
      if (selected >= 2) {
        selected = 1;
      }
      if (selected < 0) {
        selected = 0;
      }
    }
    frontEnd.language = languages[selected];
    purgememadr(shapeFile);
    Quick_DD(1,1,1);
    systemtask();
  }
  return;
}
